from flask import Flask, jsonify
from flask_cors import CORS
import requests

app = Flask(__name__)
CORS(app)

@app.route('/')
def home():
    return "Server Running"

@app.route('/data')
def data():
    url = "https://api.data.gov.in/resource/9ef84268-d588-465a-a308-a864a43d0070?format=json&api-key=YOUR_API_KEY"
    res = requests.get(url)
    return jsonify(res.json())

if __name__ == "__main__":
    app.run(debug=True)
